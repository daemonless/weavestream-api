## Deployment

### Podman Compose

```yaml
services:
  weavestream-api:
    image: "ghcr.io/daemonless/weavestream-api:latest"
    container_name: weavestream-api
    environment:
      - PUID=1000  # User ID for the API process
      - PGID=1000  # Group ID for the API process
      - TZ=UTC  # Timezone for the container
      - DATABASE_URL=postgresql://weavestream:REPLACEME@postgres:5432/weavestream  # PostgreSQL connection string (e.g. postgresql://user:pass@host:5432/db)
      - REDIS_URL=redis://:REPLACEME@redis:6379/0  # Redis or Valkey connection string (e.g. redis://host:6379/0)
      - APP_URL=http://localhost:3000  # Public origin of the web UI (default: http://localhost:3000)
      - API_URL=http://localhost:3000/api  # Public origin of the API (default: http://localhost:3000/api)
      # - JWT_SIGNING_KEY=  # 32-byte base64 secret (auto-generated if unset)
      # - MFA_ENCRYPTION_KEY=  # 32-byte base64 secret (auto-generated if unset)
      # - PASSWORD_ENCRYPTION_KEY=  # 32-byte base64 secret (auto-generated if unset)
      # - INTEGRATION_SECRET_KEY=  # 32-byte base64 secret (auto-generated if unset)
      # - SMTP_SECRET_KEY=  # 32-byte base64 secret (auto-generated if unset)
      # - COOKIE_SIGNING_KEY=  # 32-byte base64 secret (auto-generated if unset)
      # - CSRF_SIGNING_KEY=  # 32-byte base64 secret (auto-generated if unset)
    volumes:
      - "/containers/weavestream-api/data/files:/var/lib/weavestream/files"
      - "/containers/weavestream-api/data/backup:/var/lib/weavestream/backup:ro"
    ports:
      - "4000:4000"
    # always (not unless-stopped) so FreeBSD's podman rc.d auto-starts it at boot
    restart: always
```

Save as `compose.yaml`, then run `podman-compose up -d`.

### AppJail Director
**.env**:

```
# .env

DIRECTOR_PROJECT=weavestream-api
PUID=1000
PGID=1000
TZ=UTC
DATABASE_URL=postgresql://weavestream:REPLACEME@postgres:5432/weavestream
REDIS_URL=redis://:REPLACEME@redis:6379/0
APP_URL=http://localhost:3000
API_URL=http://localhost:3000/api
```

**appjail-director.yml**:

```yaml
# appjail-director.yml

options:
  - alias:
  - ip4_inherit:
services:
  weavestream-api:
    name: weavestream_api
    options:
      - container: 'args:--pull'
    oci:
      user: root
      environment:
        - PUID: !ENV '${PUID}'
        - PGID: !ENV '${PGID}'
        - TZ: !ENV '${TZ}'
        - DATABASE_URL: !ENV '${DATABASE_URL}'
        - REDIS_URL: !ENV '${REDIS_URL}'
        - APP_URL: !ENV '${APP_URL}'
        - API_URL: !ENV '${API_URL}'
    volumes:
      - weavestream-api_data_files: /var/lib/weavestream/files
      - weavestream-api_data_backup: /var/lib/weavestream/backup
  weavestream-postgres:
    name: weavestream_postgres
    options:
      - from: ghcr.io/daemonless/postgres:16
      - template: !ENV '${PWD}/postgres-template.conf'
  weavestream-redis:
    name: weavestream_redis
    options:
      - from: ghcr.io/daemonless/redis:latest
      - template: !ENV '${PWD}/template.conf'
volumes:
  weavestream-api_data_files:
    device: '/containers/weavestream-api/data/files'
  weavestream-api_data_backup:
    device: '/containers/weavestream-api/data/backup'
```

**Makejail**:

```
# Makejail

ARG tag=latest

OPTION container=boot
OPTION overwrite=force
OPTION from=ghcr.io/daemonless/weavestream-api:${tag}
```

Save the files above, then run `appjail-director up`.


> [!WARNING]
> Exposing ports in AppJail means that your service can be reached from remote hosts. If that is not your intention, do not expose the ports and communicate with the service using the jail's IPv4 address or hostname assigned by the virtual network.
>
> To avoid exposing ports, just remove the `expose` option in your `appjail-director.yml` or from your command-line arguments.

### Bastille

> [!WARNING]
